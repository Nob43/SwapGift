import asyncio import logging from aiohttp import web from aiogram import Bot, Dispatcher from app.api_handlers import cors_middleware, ddos_guard_middleware, register_api_routes from app.bot_handlers import register_bot_handlers from app.config import API_HOST, API_PORT, API_TOKEN, validate_required_settings from app.database import ( add_gift_to_db, get_user_gift_rows, get_warehouse_stars_balance, init_db, save_warehouse_gift_event, withdraw_user_gift, ) from app.warehouse import WarehouseAccount logging.basicConfig(level=logging.INFO) validate_required_settings() bot = Bot(token=API_TOKEN) dp = Dispatcher() warehouse = WarehouseAccount( bot=bot, add_gift_to_db=add_gift_to_db, save_warehouse_gift_event=save_warehouse_gift_event, ) register_bot_handlers( dp=dp, bot=bot, warehouse=warehouse, add_gift_to_db=add_gift_to_db, get_user_gift_rows=get_user_gift_rows, get_warehouse_stars_balance=get_warehouse_stars_balance, withdraw_user_gift=withdraw_user_gift, ) async def main(): init_db() await warehouse.setup() app = web.Application(middlewares=[ddos_guard_middleware, cors_middleware], client_max_size=2 * 1024 * 1024) register_api_routes(app, warehouse, bot) app.router.add_static("/static/", path="static", name="static") runner = web.AppRunner(app) await runner.setup() site = web.TCPSite(runner, API_HOST, API_PORT) await site.start() print(f"Bot and API started. API: http://{API_HOST}:{API_PORT}") while True: try: await dp.start_polling(bot) except asyncio.CancelledError: raise except Exception as exc: logging.exception("Polling stopped due to error, restarting in 3s: %s", exc) await asyncio.sleep(3) if __name__ == "__main__": asyncio.run(main())
